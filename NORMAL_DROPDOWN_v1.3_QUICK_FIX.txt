    display: flex;          ← For alignment
}

Child Dropdown:
───────────────
.lakum-nav__dropdown {
    position: absolute;     ← Position relative to parent
    top: 100%;              ← Below the parent
    left: 0;                ← Align left (LTR)
    right: auto;
    
    width: 200px;           ← Fixed width
    opacity: 0;             ← Hidden
    visibility: hidden;
    pointer-events: none;
    
    transition: opacity 0.3s;  ← Smooth fade
}

Active State:
─────────────
.lakum-nav__item--dropdown.active .lakum-nav__dropdown {
    opacity: 1;             ← Show
    visibility: visible;
    pointer-events: auto;   ← Allow clicks
}

RTL Override:
──────────────
[dir="rtl"] .lakum-nav__dropdown {
    left: auto;             ← Remove left
    right: 0;               ← Align right instead
}
