CSS CHANGES MADE:
═══════════════════════════════════════════════════════════════════════════════

BEFORE (Hidden by default):
  .lakum-nav__dropdown {
      opacity: 0;                    ← HIDDEN
      visibility: hidden;            ← HIDDEN
      transform: translateY(-10px);  ← Slides up
      pointer-events: none;          ← Can't click
  }

  .lakum-nav__item--dropdown:hover .lakum-nav__dropdown {
      opacity: 1;       ← Only visible on hover
      visibility: visible;
      transform: translateY(0);
      pointer-events: auto;
  }

AFTER (Always visible):
  .lakum-nav__dropdown {
      opacity: 1 !important;         ← ALWAYS VISIBLE
      visibility: visible !important; ← ALWAYS VISIBLE
      pointer-events: auto !important; ← Always clickable
      left: 50%;
      transform: translateX(-50%);   ← Centered below item
  }

  .lakum-nav__item--dropdown:hover .lakum-nav__dropdown-toggle {
      transform: rotate(180deg);     ← Just rotate arrow
  }
