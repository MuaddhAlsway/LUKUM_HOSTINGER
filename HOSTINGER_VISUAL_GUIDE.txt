┌─ HOW HOSTINGER WORKS ─────────────────────────────────────────────────────────┐
│                                                                                 │
│  Traditional Hosting (with .htaccess):                                         │
│  ─────────────────────────────────────────────────────────────────────────     │
│  URL: /event/dior-exhibition                                                  │
│  ↓                                                                              │
│  .htaccess rewrites to: event.php?slug=dior-exhibition                         │
│  ↓                                                                              │
│  Loads event.php                                                               │
│                                                                                 │
│  ─────────────────────────────────────────────────────────────────────────     │
│  Hostinger (File-Based):                                                       │
│  ─────────────────────────────────────────────────────────────────────────     │
│  URL: /event/dior-exhibition                                                  │
│  ↓                                                                              │
│  Hostinger looks for: /event/dior-exhibition.php                              │
│  ↓                                                                              │
│  If found, loads the file                                                      │
│  ↓                                                                              │
│  If not found, 404 error                                                       │
│                                                                                 │
│  ⚠️  NO .HTACCESS REWRITING ON HOSTINGER!                                      │
│                                                                                 │
└─────────────────────────────────────────────────────────────────────────────────┘

┌─ OUR SOLUTION: CREATE PHYSICAL FILES ──────────────────────────────────────────┐

│  ↓                                                                              │
│  Hostinger looks for: /event/dior-exhibition.php                              │
│  ↓                                                                              │
│  Finds the file (we created it!)                                              │
│  ↓                                                                              │
│  File includes: event.php                                                      │
│  ↓                                                                              │
│  event.php reads slug from filename                                           │

│  Fetches event data from database                                             │

┌─ FILE GENERATOR CODE ─────────────────────────────────────────────────────────┐
│                                                                                 │
│  class FileGenerator {                                                         │
│      public static function generateEventFile($slug) {                        │
│          $dir = __DIR__ . '/../event';                                        │
│          if (!is_dir($dir)) mkdir($dir, 0755, true);                          │
│                                                                                 │
│          $file = $dir . '/' . $slug . '.php';                                 │
│          $code = '<?php require_once __DIR__ . "/../event.php"; ?>';          │
│                                                                                 │
│          file_put_contents($file, $code);                                     │
│          chmod($file, 0644);                                                  │
│      }                                                                         │
│  }                                                                             │
│                                                                                 │
│  Usage:                                                                        │
│  FileGenerator::generateEventFile('dior-exhibition');                         │
│  → Creates: /event/dior-exhibition.php                                        │
│                                                                                 │
└─────────────────────────────────────────────────────────────────────────────────┘
