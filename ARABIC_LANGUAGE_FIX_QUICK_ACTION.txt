WHAT WAS WRONG
═══════════════════════════════════════════════════════════════════════════════

BEFORE (BROKEN):
  document.addEventListener('DOMContentLoaded', function() {
      loadLegalPageContent();  ❌ Function doesn't exist yet!
  });
  
  async function loadLegalPageContent() {
      // Function defined AFTER it's called
  }

AFTER (FIXED):
  async function loadLegalPageContent() {
      // Function defined FIRST
  }
  
  document.addEventListener('DOMContentLoaded', function() {
      loadLegalPageContent();  ✅ Function exists now!
  });
